#Relief funds
country_event = {
	id = hansa_puppet.1
	immediate = {log = "[GetDateText]: [Root.GetName]: event hansa_puppet.1"}
	title = hansa_puppet.1.t
	desc = hansa_puppet.1.d
	picture = GFX_event_solarempire

	is_triggered_only = yes

	#Fund it fully
	option = {
		name = hansa_puppet.1.a
		trigger = {
			check_variable = { var = bank_million value > 3.99 }
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				check_variable = { var = bank_million value > 19.99 }
			}
		}
		custom_effect_tooltip = NID_pay_4_mil_tt
		hidden_effect = {
			subtract_from_variable = { var = bank_million value = 4 }
		}
		FROM = {
			add_stability = 0.10
			if = {
				limit = { has_idea = NID_hp_neglected_idea }
				remove_ideas = NID_hp_neglected_idea
			}
		}
	}
	#Fund part of it
	option = {
		name = hansa_puppet.1.b
		trigger = {
			check_variable = { var = bank_million value > 1.99 }
		}
		ai_chance = {
			base = 0
			modifier = {
				add = 100
				check_variable = { var = bank_million value > 7.99 }
			}
			modifier = {
				factor = 0
				check_variable = { var = bank_million value > 19.99 }
			}
		}
		custom_effect_tooltip = NID_pay_2_mil_tt
		hidden_effect = {
			subtract_from_variable = { var = bank_million value = 2 }
		}
		FROM = {
			add_stability = 0.05
		}
	}
	#Refuse
	option = {
		name = hansa_puppet.1.c
		ai_chance = {
			base = 1
		}
		FROM = {
			add_timed_idea = { idea = NID_hp_neglected_idea days = 180 }
		}
	}
}

#Loads the Hansa puppet tree (fired by NID_load_hansa_puppet_tree)
country_event = {
	id = hansa_puppet.10
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event hansa_puppet.10"
		if = {
			limit = {
				is_subject_of = NID
				NOT = { has_focus_tree = NID_hansa_puppet_tree }
			}
			load_focus_tree = NID_hansa_puppet_tree
			mark_focus_tree_layout_dirty = yes
		}
	}
}

#Shareholder report loop on the puppet (started by NID_hp_shareholder_report, yearly)
country_event = {
	id = hansa_puppet.20
	hidden = yes
	is_triggered_only = yes

	immediate = {
		log = "[GetDateText]: [Root.GetName]: event hansa_puppet.20"
		if = {
			limit = { is_subject_of = NID }
			NID = { country_event = { id = hansa_puppet.21 days = 1 } }
			country_event = { id = hansa_puppet.20 days = 365 }
		}
	}
}
