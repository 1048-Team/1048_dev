#Hansa puppet tree for NID subjects
on_actions = {
	on_puppet = {
		effect = {
			if = {
				limit = { FROM = { tag = NID } }
				ROOT = { NID_load_hansa_puppet_tree = yes }
			}
		}
	}
	on_release_as_puppet = {
		effect = {
			if = {
				limit = { FROM = { tag = NID } }
				ROOT = { NID_load_hansa_puppet_tree = yes }
			}
		}
	}
}
