griffonia_calm_down_a_bit = {
    IF = {
        limit = {
            OR = {
                tag = EQC
                tag = GRI
            }
        }
        IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 1 }
            }
            add_threat = -42.5
        }
        ELSE_IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 2 }
            }
            add_threat = -30
        }
        ELSE_IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 3 }
            }
            add_threat = -15
        }
    }
    ELSE_IF = {
        limit = {
            tag = BRF
        }
        IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 1 }
            }
            add_threat = -15
        }
        ELSE_IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 2 }
            }
            add_threat = -8
        }
        ELSE_IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 3 }
            }
            add_threat = -4
        }
    }
    ELSE_IF = {
        limit = {
            tag = RCT
        }
        IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 1 }
            }
            add_threat = -7.4
        }
        ELSE_IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 2 }
            }
            add_threat = -3
        }
        IF = {
            limit = {
                GRI = {
                    is_ai = no
                }
            }
            IF = {
            limit = {
                check_variable = { FH_limited_war_escalator = 1 }
            }
                add_threat = -11
            }
            ELSE_IF = {
                limit = {
                    check_variable = { FH_limited_war_escalator = 2 }
                }
                add_threat = -4
            }
        }
    }
}
