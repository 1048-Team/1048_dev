    EQC_GRI_war_countdown = {
        activation = {
            check_variable = { global.FH_limited_war_escalator < 4 }
            has_war_with = GRI
            hidden_trigger = {
                OR = {
                    tag = EQC
                    is_in_faction_with = EQC
                }
            }
        }
        available = {
            owns_needed_GRI_territory_EQC = yes
            if = {
                limit = {
                    is_subject_of = EQC
                }
                NOT = { is_subject_of = EQC }
            }
            else_if = {
                limit = {
                    is_in_faction_with = EQC
                    NOT = { tag = EQC }
                }
                NOT = { is_in_faction_with = EQC }
            }
        }
        days_mission_timeout = 100
        complete_effect = {
            log = "[GetDateText]: [Root.GetName]: mission complete EQC_GRI_war_countdown"
            custom_effect_tooltip = griffonia_expand_influence_tt
            hidden_effect = {
                if = {
                    limit = {
                        tag = EQC
                    }
                    griffonia_GRI_land_transfer_EQC = yes
                    add_to_variable = { global.griffonia_world_tension_tracker = 1 }
                }
            }
        }
        timeout_effect = {
            log = "[GetDateText]: [Root.GetName]: mission timeout EQC_GRI_war_countdown"
            custom_effect_tooltip = griffonia_wimp_out_tt
            hidden_effect = {
                if = {
                    limit = {
                        tag = EQC
                    }
                    griffonia_peace_out_GRI_faction = yes
                    griffonia_peace_out_RCT_faction = yes
                    add_to_variable = { global.griffonia_world_tension_tracker = 1 }
                }
            }
        }
